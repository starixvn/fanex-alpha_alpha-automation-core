	and T.Sporttype between 180 and 186

------------------------------ Update PT DCS --Begin----------------------------------

SELECT @AgentID = recommend
	FROM dbo.custInfo WITH (NOLOCK)
	WHERE custid in (select custid from #QC_Customer_SE where roleid=1)
	
-- update SMA PT
update T with (rowlock, updlock)
	set SuperPT = q.s, MasterPT = q.m, AgentPT = q.a, AutoMasterPT = q.um, IsAutoMasterPT = q.im
		,AutoSuperPT = q.u, IsAutoSuperPT = q.i, MinPT=q.n,MaxPT = 90
	From CustPTSetting3rd_Agent T 
	join Customer c with (nolock) on t.custid = c.custid
	join QC_User_PT_TEST q with (nolock) on   c.roleid = q.roleid and q.Sporttype = T.sporttype
	join #QC_Customer_SE cse with (nolock) on cse.CustID = c.custid and cse.groupid  = q.groupid 
	where q.Sporttype in (select DISTINCT sporttype from ProductSettingDetail with (NOLOCK) where SettingType=1)

-- update group PT	
update T with (rowlock, updlock)
	set AgentPT = q.a, MasterPT = q.m, SuperPT = q.s, AutoMasterPT = q.um, IsAutoMasterPT = q.im
		,AutoSuperPT = q.u, IsAutoSuperPT = q.i
	From CustPTGroupSetting3rd_Member T 
	join Customer c with (nolock) on t.AgentID = c.recommend
	join QC_User_PT_TEST q with (nolock) on   c.roleid = q.roleid and q.roleid=1 and q.Sporttype = T.sporttype
	join #QC_Customer_SE cse with (nolock) on cse.CustID = c.custid and cse.groupid  = q.groupid 
	where T.AgentID=@AgentID and q.Sporttype in (select DISTINCT sporttype from ProductSettingDetail with (NOLOCK) where SettingType=1)
	
	
-- update individual PT
update T with (rowlock, updlock)
	set SuperPT = q.s, MasterPT = q.m, AgentPT = q.a, AutoMasterPT = q.um, IsAutoMasterPT = q.im
		,AutoSuperPT = q.u, IsAutoSuperPT = q.i
	From CustPTSetting3rd_Member T 
	join Customer c with (nolock) on t.custid = c.custid
	join QC_User_PT_TEST q with (nolock) on   c.roleid = q.roleid and q.Sporttype = T.sporttype
	join #QC_Customer_SE cse with (nolock) on cse.CustID = c.custid and cse.groupid  = q.groupid
