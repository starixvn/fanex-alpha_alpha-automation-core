    [Arguments]    ${transId}
    [Documentation]    This action to get icon Event status of ticket by input TransId and return a string with format like database
    ...
    ...    Ex: 1FL:1-1;FL:1-2;
    ## Set variable
    ${homeLocator}    Get Locator by TransId    ${bl_lblChoiceMatchHome_microsite}    ${transId}
    ${awayLocator}    Get Locator by TransId    ${bl_lblChoiceMatchAway_microsite}    ${transId}
    ${VsLocator}    Get Locator by TransId    ${bl_lblChoiceMatchVS_microsite}    ${transId}
    ${divIndex}    set variable    1
    ${ac_FL_Status}    Set Variable    FL:FirstGoal-LastGoal;
    ${ac_1FL_Status}    Set Variable    1FL:1stFirstGoal-1stLastGoal;
    ${EventStatus}    Set Variable    ${EMPTY}
    # GET HOME STATUS
    Comment    # Haft Last Ball
    ${isExits}    Run Keyword And Return Status    Element Should Be Visible    ${VsLocator}/preceding-sibling::div[${divIndex}][@class='fhLastGoal']
    Run Keyword And Return If    '${isExits}'=='False'    Set Variable    ${EMPTY}
    ${is_1st_HomeLastGoal}    Run Keyword And Return Status    Get Text    ${VsLocator}/preceding-sibling::div[${divIndex}][@class='fhLastGoal']
    ${ac_1FL_Status}=    Run Keyword If    ${is_1st_HomeLastGoal}    Replace String    ${ac_1FL_Status}    1stLastGoal    1
    ...    ELSE    Set Variable    ${ac_1FL_Status}
    ${divIndex}    Run Keyword If    ${is_1st_HomeLastGoal}    evaluate    ${divIndex}+1
    ...    ELSE    Set Variable    ${divIndex}
    Comment    # Haft First Ball
    ${is_1st_HomeFirstGoal}    Run Keyword And Return Status    Get Text    ${VsLocator}/preceding-sibling::div[${divIndex}][@class='fhFirstGoal']
    ${ac_1FL_Status}=    Run Keyword If    ${is_1st_HomeFirstGoal}    Replace String    ${ac_1FL_Status}    1stFirstGoal    1
    ...    ELSE    Set Variable    ${ac_1FL_Status}

    ...    ELSE    Set Variable    ${divIndex}
    Comment    # Last Ball
    ${isHomeLastGoal}    Run Keyword And Return Status    Get Text    ${VsLocator}/preceding-sibling::div[${divIndex}][@class='lastGoal']
    ${ac_FL_Status}=    Run Keyword If    ${isHomeLastGoal}    Replace String    ${ac_FL_Status}    LastGoal    1
    ...    ELSE    Set Variable    ${ac_FL_Status}
    ${divIndex}    Run Keyword If    ${isHomeLastGoal}    evaluate    ${divIndex}+1
    ...    ELSE    Set Variable    ${divIndex}
    Comment    # First Ball
    ${isHomeFirstGoal}    Run Keyword And Return Status    Get Text    ${VsLocator}/preceding-sibling::div[${divIndex}][@class='firstGoal ']
    ${ac_FL_Status}=    Run Keyword If    ${isHomeFirstGoal}    Replace String    ${ac_FL_Status}    FirstGoal    1
    ...    ELSE    Set Variable    ${ac_FL_Status}
    # GET AwaySTATUS
    ${divIndex}    set variable    1
    Comment    # First Ball
    ${isAwayFirstGoal}    Run Keyword And Return Status    Get Text    ${awayLocator}/following-sibling::div[${divIndex}][@class='firstGoal ']
    ${ac_FL_Status}=    Run Keyword If    ${isAwayFirstGoal}    Replace String    ${ac_FL_Status}    FirstGoal    2
    ...    ELSE    Set Variable    ${ac_FL_Status}
    ${divIndex}    Run Keyword If    ${isAwayFirstGoal}    evaluate    ${divIndex}+1
    ...    ELSE    Set Variable    ${divIndex}
    Comment    # Last Ball
    ${isAwayLastGoal}    Run Keyword And Return Status    Get Text    ${awayLocator}/following-sibling::div[${divIndex}][@class='lastGoal']
    ${ac_FL_Status}=    Run Keyword If    ${isAwayLastGoal}    Replace String    ${ac_FL_Status}    LastGoal    2
    ...    ELSE    Set Variable    ${ac_FL_Status}
    ${divIndex}    Run Keyword If    ${isAwayLastGoal}    evaluate    ${divIndex}+1
    ...    ELSE    Set Variable    ${divIndex}
    Comment    # Haft First Ball
    ${is_1st_AwayFirstGoal}    Run Keyword And Return Status    Get Text    ${awayLocator}/following-sibling::div[${divIndex}][@class='fhFirstGoal']
    ${ac_1FL_Status}=    Run Keyword If    ${is_1st_AwayFirstGoal}    Replace String    ${ac_1FL_Status}    1stFirstGoal    2
    ...    ELSE    Set Variable    ${ac_1FL_Status}
    ${divIndex}    Run Keyword If    ${is_1st_AwayFirstGoal}    evaluate    ${divIndex}+1
    ...    ELSE    Set Variable    ${divIndex}
    Comment    # Haft Last Ball
    ${is_1st_AwayLastGoal}    Run Keyword And Return Status    Get Text    ${awayLocator}/following-sibling::div[${divIndex}][@class='fhLastGoal']
    ${ac_1FL_Status}=    Run Keyword If    ${is_1st_AwayLastGoal}    Replace String    ${ac_1FL_Status}    1stLastGoal    2
    ...    ELSE    Set Variable    ${ac_1FL_Status}
    # Set 0 if NoGoal
    ${ac_FL_Status}=    Replace String    ${ac_FL_Status}    FirstGoal    0
    ${ac_FL_Status}=    Replace String    ${ac_FL_Status}    LastGoal    0
    ${ac_1FL_Status}=    Replace String    ${ac_1FL_Status}    1stFirstGoal    0
    ${ac_1FL_Status}=    Replace String    ${ac_1FL_Status}    1stLastGoal    0
